# Tooling for the ball tracking spike. Mac-side only.
#
# Build it:
#
#   python3 -m venv tools/.venv
#   ./tools/.venv/bin/python -m pip install -r tools/requirements.txt
#
# Everything in tools/ is then run through ./tools/.venv/bin/python. Known
# working on Python 3.14.4, which is what this project uses; torch and
# opencv wheels both resolved there without needing an older interpreter.
# If a future Python is too new for a wheel, drop back one minor version
# rather than waiting.
#
# THIS IS NOT THE EXPORT ENVIRONMENT. Core ML conversion needs older, pinned
# versions and lives in tools/.venv-export -- see tools/requirements-export.txt
# for why the two are deliberately separate.
#
# This is all deliberately separate from the app, which has no third-party
# dependencies and is not affected by anything here. These files live outside
# GoalKick/ so Xcode's synchronized folders do not sweep them into the target.

opencv-python>=4.10
numpy>=1.26

# Pre-trained object detection. COCO's "sports ball" class finds footballs of
# any colour without training data, which a colour threshold cannot do.
#
# This pulls in PyTorch, the dependency most likely to lack a wheel for a
# very new Python. On macOS arm64 the wheel is around 65 MB, not the multi-
# gigabyte download the CUDA builds require.
ultralytics>=8.3
